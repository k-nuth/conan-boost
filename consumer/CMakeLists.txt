cmake_minimum_required(VERSION 3.25)
project(boost_consumer LANGUAGES CXX)
find_package(Boost CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(boost_consumer main.cpp)
target_compile_features(boost_consumer PRIVATE cxx_std_23)
target_compile_definitions(boost_consumer PRIVATE EXPECTED_BOOST_VERSION=${EXPECTED_BOOST_VERSION})
target_link_libraries(boost_consumer PRIVATE Boost::headers Boost::boost Threads::Threads)
if(TEST_NATIVE)
    find_package(Boost CONFIG REQUIRED COMPONENTS thread)
    target_compile_definitions(boost_consumer PRIVATE TEST_NATIVE)
    target_link_libraries(boost_consumer PRIVATE Boost::thread)
endif()
